    Private Function IsHeaderRow( _
    ByVal ws As Object, _
    ByVal rowNumber As Long) As Boolean

    Dim col1 As String
    Dim col2 As String

    col1 = Trim$(GetCellText(ws, rowNumber, 1))
    col2 = Trim$(GetCellText(ws, rowNumber, 2))

    IsHeaderRow = _
        (col1 = "—œÌ›" And _
        InStr(1, col2, "‘„«—Â »Ì„Â", vbTextCompare) > 0)

End Function


Private Function IsEmptyReportRow( _
    ByVal ws As Object, _
    ByVal rowNumber As Long) As Boolean

    Dim i As Integer

    For i = 1 To 22

        If Trim$(GetCellText(ws, rowNumber, i)) <> "" Then
            IsEmptyReportRow = False
            Exit Function
        End If

    Next i

    IsEmptyReportRow = True

End Function


Private Function GetCellText( _
    ByVal ws As Object, _
    ByVal rowNumber As Long, _
    ByVal columnNumber As Long) As String

    Dim v As Variant

    v = ws.Cells(rowNumber, columnNumber).Value2

    If IsError(v) Or IsNull(v) Or IsEmpty(v) Then
        GetCellText = ""
    Else
        GetCellText = Trim$(CStr(v))
    End If

End Function


Private Sub SetTextField( _
    ByRef rs As DAO.Recordset, _
    ByVal fieldName As String, _
    ByVal fieldValue As Variant)

    If IsNull(fieldValue) Then

        rs.fields(fieldName).value = Null

    ElseIf Trim$(CStr(fieldValue)) = "" Then

        rs.fields(fieldName).value = Null

    Else

        rs.fields(fieldName).value = CStr(fieldValue)

    End If

End Sub


Private Sub SetLongField( _
    ByRef rs As DAO.Recordset, _
    ByVal fieldName As String, _
    ByVal fieldValue As Variant)

    If IsNumeric(fieldValue) Then
        rs.fields(fieldName).value = CLng(fieldValue)
    Else
        rs.fields(fieldName).value = Null
    End If

End Sub


Private Sub SetCurrencyField( _
    ByRef rs As DAO.Recordset, _
    ByVal fieldName As String, _
    ByVal fieldValue As Variant)

    If IsNumeric(fieldValue) Then
        rs.fields(fieldName).value = CCur(fieldValue)
    Else
        rs.fields(fieldName).value = Null
    End If

End Sub
